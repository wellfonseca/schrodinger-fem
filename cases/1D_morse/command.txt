# =============================================================================
#  CASO: 1D_morse   (Elmer FEM)   --  1D
#  L = 10.0   n = 400   x0 = 0.0
#  V = V = 10(e^-2b(x-3) - 2e^-b(x-3))
#  Analitico: minimo = -De
#  Esperado:  V_min = -10, E0 = -8.481
# =============================================================================
#
#  IMPORTANTE: rode SEMPRE nos nos computacionais, NUNCA no headnode.
#  Use 'qsub' (recomendado) ou 'qsub -I' para sessao interativa num no.
#
# =============================================================================
#  PASSO 0 -- entrar no cluster e carregar modulos
# =============================================================================
ssh wfonseca@10.107.1.23
source /etc/profile.d/modules.sh
module load gnu12 openmpi4 openblas
export LC_ALL=C

# =============================================================================
#  PASSO 1 -- variaveis de ambiente do Elmer (COPIAR E COLAR)
# =============================================================================
export ELMER_INSTALL=$HOME/elmer_build/install_mumps
export ELMER_HOME=$ELMER_INSTALL
export ELMER_SOLVER_HOME=$ELMER_INSTALL/share/elmersolver
export MUMPSROOT=/opt/ohpc/pub/libs/gnu12/openmpi4/mumps/5.2.1
export SCALAPACK_ROOT=/opt/ohpc/pub/libs/gnu12/openmpi4/scalapack/2.2.0
export LD_LIBRARY_PATH=$ELMER_INSTALL/lib/elmersolver:$MUMPSROOT/lib:$SCALAPACK_ROOT/lib:$LD_LIBRARY_PATH
export ES=$ELMER_INSTALL/bin/ElmerSolver

# =============================================================================
#  PASSO 2 -- ir para a pasta do caso
# =============================================================================
cd ~/P1_2_casos/casos/1D_morse

# =============================================================================
#  PASSO 3 -- gerar a malha no formato do Elmer
#             (cria ./malha/ com mesh.header, mesh.nodes, mesh.elements, mesh.boundary)
# =============================================================================
python3 gerar_malha.py

# =============================================================================
#  PASSO 4 -- rodar o solver
# =============================================================================
$ES caso.sif > run.log 2>&1
# equivalente ao exemplo:  $ ElmerSolver

# =============================================================================
#  PASSO 5 -- ver os autovalores (energias)
# =============================================================================
grep "E(" run.log
# esperado: V_min = -10, E0 = -8.481

# =============================================================================
#  PASSO 6 -- abrir no ParaView
# =============================================================================
paraview malha/run_t0001.vtu
#
#  O arquivo .vtu e gerado pelo ResultOutputSolver (Solver 2 no caso.sif).
#  Ele contem a malha + o campo 'psi' (autofuncao do modo 1).
#  No ParaView: escolha 'psi' em Coloring para ver a autofuncao.

# =============================================================================
#  ALTERNATIVA -- tudo de uma vez via PBS (recomendado)
# =============================================================================
#   Na sua maquina local:
#     cd schrodinger/01_simulacoes/schrodinger_P1_2/casos/1D_morse

# =============================================================================
#  FLUXO COMANDOS PADRAO (GMSH + ElmerGrid + ElmerSolver)
# =============================================================================
#  Este e o fluxo recomendado: malha gerada no GMSH a partir do .geo que ja
#  esta nesta pasta, convertida pelo ElmerGrid e resolvida pelo ElmerSolver.
#
#  --- 0) ambiente ---
#  source ../ambiente_local.sh        # deixa os comandos no PATH e resolve o MKL
#
#  --- 1) GMSH: .geo -> .msh ---
#  gmsh 1D_morse.geo -2 -o 1D_morse.msh
#  (use -3 em vez de -2 para os casos 3D)
#
#  --- 2) ElmerGrid: .msh -> mesh/ ---
#  ElmerGrid 14 2 1D_morse.msh -out mesh
#  #  14 = entrada GMSH   |   2 = saida ElmerSolver
#  #  ATENCAO: 4 e Ansys, NAO Gmsh.
#
#  --- 3) ElmerSolver ---
#  ElmerSolver caso.sif
#  # ou, em paralelo (ver o aviso no LEIA-ME):
#  mpirun -np 4 ElmerSolver_mpi caso.sif
#
#  --- 4) resultados ---
#  grep "E(" run.log
#
#  --- 5) ParaView ---
#  paraview mesh/run_t0001.vtu
#  #  o Elmer escreve no diretorio da MALHA, por isso 'mesh/'
#
#  --- TUDO DE UMA VEZ (script) ---
#  bash ../gera_malha_gmsh.sh 1D_morse
#  ElmerSolver caso.sif
# =============================================================================

# =============================================================================
#  SE ALGO FALHAR
# =============================================================================
# 1) "cannot open shared object file" -> confira o PASSO 1 (LD_LIBRARY_PATH)
# 2) "Requested mesh ... does not exist" -> rode o PASSO 3 (gerar_malha.py)
# 3) sem .vtu -> confirme que caso.sif tem o Solver 2 (ResultOutputSolver)
# 4) autovalores vazios -> veja os erros:
#       grep -iE "ERROR|Fatal" run.log | grep -v LUA
